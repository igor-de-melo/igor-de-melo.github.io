import numpy as np

def forward(X, W1, b1, W2, b2):
    h = np.tanh(X @ W1 + b1)
    return h, h @ W2 + b2

def train_step(X, y, W1, b1, W2, b2, lr=1e-2):
    h, y_hat = forward(X, W1, b1, W2, b2)
    err = y_hat - y
    dW2 = h.T @ err / len(X)
    dh = (err @ W2.T) * (1 - h**2)
    dW1 = X.T @ dh / len(X)
    W1 -= lr * dW1; b1 -= lr * dh.mean(0)
    W2 -= lr * dW2; b2 -= lr * err.mean(0)
    return (err**2).mean()
